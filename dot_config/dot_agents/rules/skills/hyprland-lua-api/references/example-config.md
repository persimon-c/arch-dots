# Hyprland Lua — Annotated Example Config

Source: https://github.com/hyprwm/Hyprland/blob/main/example/hyprland.lua

```lua
-- ~/.config/hypr/hyprland.lua
-- Split config across files with: require("myColors")

------------------ MONITORS ------------------
hl.monitor({
  output   = "",          -- "" matches all outputs
  mode     = "preferred",
  position = "auto",
  scale    = "auto",
})

------------------ PROGRAMS ------------------
local terminal    = "kitty"
local fileManager = "dolphin"
local menu        = "hyprlauncher"

------------------ AUTOSTART ------------------
-- Use hyprland.start to avoid re-running on config reload
hl.on("hyprland.start", function()
  hl.dispatch(hl.dsp.exec_cmd("waybar"))
  hl.dispatch(hl.dsp.exec_cmd("hyprpaper"))
  hl.dispatch(hl.dsp.exec_cmd("hypridle"))
end)

------------------ ENV VARS ------------------
-- Note: with uwsm, put env vars in ~/.config/uwsm/env instead
hl.env("XCURSOR_SIZE", "24")
hl.env("HYPRCURSOR_SIZE", "24")

------------------ CONFIG ------------------
hl.config({
  autogenerated = false,   -- removes yellow warning banner

  general = {
    gaps_in  = 5,
    gaps_out = 20,
    border_size = 2,
    col = {
      active_border   = "rgba(33ccffee) rgba(00ff99ee) 45deg",
      inactive_border = "rgba(595959aa)",
    },
    resize_on_border = false,
    allow_tearing    = false,
    layout           = "dwindle",
  },

  decoration = {
    rounding       = 10,
    rounding_power = 2,
    active_opacity   = 1.0,
    inactive_opacity = 1.0,
    shadow = {
      enabled      = true,
      range        = 4,
      render_power = 3,
      color        = "rgba(1a1a1aee)",
    },
    blur = {
      enabled   = true,
      size      = 3,
      passes    = 1,
      vibrancy  = 0.1696,
    },
  },

  animations = {
    enabled                = true,
    first_launch_animation = true,
  },

  dwindle = {
    pseudotile    = true,
    preserve_split = true,
  },

  master = {
    new_status = "master",
  },

  misc = {
    force_default_wallpaper = -1,
    disable_hyprland_logo   = false,
  },
})

------------------ ANIMATION CURVES ------------------
hl.curve("easeOutQuint",   { type = "bezier", points = { {0.23, 1},    {0.32, 1}    } })
hl.curve("easeInOutCubic", { type = "bezier", points = { {0.65, 0.05}, {0.36, 1}    } })
hl.curve("linear",         { type = "bezier", points = { {0, 0},       {1, 1}       } })
hl.curve("almostLinear",   { type = "bezier", points = { {0.5, 0.5},   {0.75, 1}    } })
hl.curve("quick",          { type = "bezier", points = { {0.15, 0},    {0.1, 1}     } })
hl.curve("easy",           { type = "spring", mass = 1, stiffness = 71.2633, dampening = 15.8273644 })

------------------ INPUT ------------------
hl.config({
  input = {
    kb_layout  = "us",
    follow_mouse = 1,
    sensitivity  = 0,
    touchpad = {
      natural_scroll = false,
    },
  },
  gestures = {
    workspace_swipe = false,
  },
})

------------------ KEYBINDS ------------------
local mainMod = "SUPER"

-- Apps
hl.bind(mainMod .. " + RETURN", hl.dsp.exec_cmd(terminal))
hl.bind(mainMod .. " + E",      hl.dsp.exec_cmd(fileManager))
hl.bind(mainMod .. " + R",      hl.dsp.exec_cmd(menu))

-- Exit (prefer hyprshutdown if available)
hl.bind(mainMod .. " + M",
  hl.dsp.exec_cmd("command -v hyprshutdown >/dev/null 2>&1 && hyprshutdown || hyprctl dispatch 'hl.dsp.exit()'"))

-- Window management
hl.bind(mainMod .. " + Q",          hl.dsp.window.close())
hl.bind(mainMod .. " + V",          hl.dsp.window.float({ action = "toggle" }))
hl.bind(mainMod .. " + F",          hl.dsp.window.fullscreen({ mode = "fullscreen" }))
hl.bind(mainMod .. " + P",          hl.dsp.window.pin())

-- Focus with HJKL or arrows
hl.bind(mainMod .. " + h",          hl.dsp.window.focus_dir({ dir = "l" }))
hl.bind(mainMod .. " + l",          hl.dsp.window.focus_dir({ dir = "r" }))
hl.bind(mainMod .. " + k",          hl.dsp.window.focus_dir({ dir = "u" }))
hl.bind(mainMod .. " + j",          hl.dsp.window.focus_dir({ dir = "d" }))

-- Move windows
hl.bind(mainMod .. " + SHIFT + h",  hl.dsp.window.move({ dir = "l" }))
hl.bind(mainMod .. " + SHIFT + l",  hl.dsp.window.move({ dir = "r" }))
hl.bind(mainMod .. " + SHIFT + k",  hl.dsp.window.move({ dir = "u" }))
hl.bind(mainMod .. " + SHIFT + j",  hl.dsp.window.move({ dir = "d" }))

-- Workspaces 1-9 + 0
for i = 1, 9 do
  hl.bind(mainMod .. " + " .. i,
    hl.dsp.workspace.focus({ workspace = tostring(i) }))
  hl.bind(mainMod .. " + SHIFT + " .. i,
    hl.dsp.window.move_to_workspace({ workspace = tostring(i) }))
end
hl.bind(mainMod .. " + 0",          hl.dsp.workspace.focus({ workspace = "10" }))
hl.bind(mainMod .. " + SHIFT + 0",  hl.dsp.window.move_to_workspace({ workspace = "10" }))

-- Scroll through workspaces with mouse
hl.bind(mainMod .. " + mouse_down", hl.dsp.workspace.scroll({ dir = "up" }))
hl.bind(mainMod .. " + mouse_up",   hl.dsp.workspace.scroll({ dir = "down" }))

-- Move/resize windows with mouse
hl.bind(mainMod .. " + mouse:272",  hl.dsp.window.move_float())    -- LMB drag
hl.bind(mainMod .. " + mouse:273",  hl.dsp.window.resize_float())  -- RMB drag

-- Special workspace (scratchpad)
hl.bind(mainMod .. " + S",          hl.dsp.workspace.toggle_special("magic"))
hl.bind(mainMod .. " + SHIFT + S",  hl.dsp.window.move_to_workspace({ workspace = "special:magic" }))

-- Dwindle layout
hl.bind(mainMod .. " + SHIFT + P",  hl.dsp.layout("dwindle:togglesplit"))

-- Resize submap
hl.bind(mainMod .. " + ALT + r", hl.dsp.submap("resize"))
hl.bind("l", hl.dsp.window.resize({ dir = "r", amount = 30 }), { submap = "resize" })
hl.bind("h", hl.dsp.window.resize({ dir = "l", amount = 30 }), { submap = "resize" })
hl.bind("k", hl.dsp.window.resize({ dir = "u", amount = 30 }), { submap = "resize" })
hl.bind("j", hl.dsp.window.resize({ dir = "d", amount = 30 }), { submap = "resize" })
hl.bind("ESCAPE", hl.dsp.submap("reset"), { submap = "resize" })

------------------ WINDOW RULES ------------------
-- Float dialogs/pickers
hl.window_rule({ match = { class = "^(org.kde.polkit-kde-authentication-agent-1)$" }, float = true })
hl.window_rule({ match = { class = "^(xdg-desktop-portal)$" },   float = true })
hl.window_rule({ match = { class = "^(file_progress)$" },         float = true })
hl.window_rule({ match = { title = "^(Open File)(.*)$" },         float = true })
hl.window_rule({ match = { title = "^(Select a File)(.*)$" },     float = true })

-- Opacity rules
hl.window_rule({ match = { class = "kitty" },   opacity = 0.95, opacity_inactive = 0.85 })

-- Workspace assignments
hl.window_rule({ match = { class = "Spotify" }, workspace = "5" })
hl.window_rule({ match = { class = "discord" }, workspace = "4" })

------------------ WORKSPACE RULES ------------------
hl.workspace_rule({ workspace = "1", monitor = "DP-1", default = true })
hl.workspace_rule({ workspace = "2", monitor = "DP-1" })
hl.workspace_rule({ workspace = "3", monitor = "HDMI-A-1", default = true })

------------------ PERMISSIONS ------------------
hl.permission({ binary = "/usr/bin/hyprlock", type = "screencopy", mode = "allow" })
```
